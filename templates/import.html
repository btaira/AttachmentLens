Import / Update Posts

1

Open Derek Hart's Facebook profile in Chrome:
facebook.com/derek.michael.hart
Scroll down until a few posts are visible.

2

Access Chrome Developer Tools — press Ctrl + Shift + I (or F12), then click the Console tab.

3

Set how many posts, copy the script, paste into the console, press Enter.

(async function() {

  // ── CONFIG ──────────────────────────────────────
  const TARGET = 20;  // posts to collect
  // ────────────────────────────────────────────────

  let collected = [], seen = new Set(), staleRounds = 0;
  const wait = ms => new Promise(r => setTimeout(r, ms));
  console.log('🚀 Starting — target: ' + TARGET + ' posts');

  async function expandSeeMore() {
    [...document.querySelectorAll('[role="button"]')]
      .filter(el => el.innerText?.trim() === 'See more')
      .forEach(btn => { try { btn.click(); } catch(e) {} });
    await wait(1500);
  }

  function harvest() {
    for (let body of document.querySelectorAll('[data-ad-comet-preview="message"]')) {
      if (collected.length >= TARGET) break;
      [...body.querySelectorAll('[role="button"]')]
        .filter(el => el.innerText?.trim() === 'See more')
        .forEach(btn => { try { btn.click(); } catch(e) {} });
      let node = body, postUrl = '', dateLabel = '';
      for (let d = 0; d < 15; d++) {
        node = node.parentElement; if (!node) break;
        const lnk = node.querySelector('a[href*="/posts/"],a[href*="story_fbid"]');
        if (lnk) { postUrl = lnk.href; dateLabel = lnk.innerText?.trim() || ''; break; }
      }
      const key = postUrl || body.innerText?.trim().substring(0, 60);
      if (seen.has(key)) continue; seen.add(key);

      // Preserve blank lines — they represent intentional pauses/spacing
      const paras = [...body.querySelectorAll('[dir="auto"]')]
        .map(el => el.innerText?.trimEnd() ?? '');
      const deduped = paras.filter((t, i) => t !== paras[i - 1] || t === '');
      const text = deduped.join('\n\n').replace(/(\n\n){3,}/g, '\n\n\n').trim();
      if (!text || text.length < 30) continue;

      // ── Likes & comments extraction ──────────────────────────────
      function parseKM(s) {
        s = (s || '').replace(/,/g, '').trim();
        if (/[Kk]$/.test(s)) return Math.round(parseFloat(s) * 1000);
        if (/[Mm]$/.test(s)) return Math.round(parseFloat(s) * 1000000);
        return parseInt(s) || 0;
      }
      let likes = 0, comments = 0;

      // Walk up to [role="article"] — Facebook's post container
      let art = body;
      for (let d = 0; d < 15; d++) {
        if (!art.parentElement) break;
        art = art.parentElement;
        if (art.getAttribute('role') === 'article') break;
      }

      // 1. Reactions (likes): Facebook splits by type in aria-labels like "Like: 56 people",
      //    "Love: 15 people", "Care: 3 people", etc. Sum them all for total reaction count.
      const REACTION_TYPES = /^(Like|Love|Care|Haha|Wow|Sad|Angry):\s*([\d,.]+[KkMm]?)\s*people/i;
      for (const el of art.querySelectorAll('[aria-label]')) {
        const lbl = (el.getAttribute('aria-label') || '').trim();
        const m = lbl.match(REACTION_TYPES);
        if (m) { likes += parseKM(m[2]); }
      }

      // 2. Engagement block: Facebook renders a short text block like "75 8 17"
      //    meaning [total-reactions, comments, shares]. Walk up, find it, take index [1].
      //    This block appears as a text node / innerText of a small container.
      //    Match: 1–5 digits, space, 1–5 digits, space, 1–5 digits (no other text).
      const ENGAGE_RE = /^(\d{1,5})\s+(\d{1,5})\s+(\d{1,5})$/;
      const allEls = art.querySelectorAll('*');
      for (const el of allEls) {
        // Only look at leaf-ish nodes whose direct text matches pattern
        const t = (el.innerText || '').trim();
        const m = t.match(ENGAGE_RE);
        if (m) {
          // m[1]=reactions, m[2]=comments, m[3]=shares
          // If we already got likes from aria-labels, trust that; otherwise use m[1]
          if (!likes) likes = parseInt(m[1]);
          comments = parseInt(m[2]);
          break;
        }
      }

      // 3. Fallback: look for "X Comments" or "X Comment" link text
      if (comments === 0) {
        for (const el of art.querySelectorAll('a, span')) {
          const t = (el.innerText || '').trim();
          if (/^\d[\d,.]*\s*[KkMm]?\s*[Cc]omments?$/.test(t)) {
            const m = t.match(/^([\d,.]*\d[KkMm]?)/);
            if (m) { comments = parseKM(m[1]); break; }
          }
        }
      }

      const popularity = likes + comments;

      collected.push({ date: dateLabel, url: postUrl, text, likes, comments, popularity });
      console.log('✅ [' + collected.length + '/' + TARGET + '] 👍' + likes + ' 💬' + comments + ' — ' + text.replace(/\n/g,' ').substring(0,60) + '...');
    }
  }

  for (let round = 0; collected.length < TARGET; round++) {
    console.log('↓ Scroll #' + (round+1) + ' — ' + collected.length + ' posts so far');
    await expandSeeMore();
    const prev = collected.length;
    harvest(); await wait(1000); harvest();
    if (collected.length === prev) {
      if (++staleRounds >= 5) { console.log('⚠️ Page exhausted.'); break; }
    } else staleRounds = 0;
    window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });
    await wait(3000);
  }

  if (!collected.length) { console.log('❌ Nothing found.'); return; }
  window.derekPosts = collected;
  console.log('\n✅ DONE — ' + collected.length + ' posts. Copying to clipboard...');
  await wait(800);

  // DevTools copy() is the most reliable method in console context.
  // Using compact JSON (no indentation) keeps the string small and fast.
  let ok = false;
  try {
    copy(JSON.stringify(window.derekPosts));
    ok = true;
    console.log('📋 Copied! Switch to the Import page, click into the JSON box, and press Ctrl+V.');
  } catch(e) {
    console.log('⚠️ copy() failed: ' + e.message);
  }
  if (!ok) {
    // Fallback: navigator.clipboard works on HTTPS (Facebook is HTTPS)
    try {
      await navigator.clipboard.writeText(JSON.stringify(window.derekPosts));
      ok = true;
      console.log('📋 Copied via clipboard API! Switch to Import page and paste (Ctrl+V).');
    } catch(e) {
      console.log('⚠️ Clipboard API failed: ' + e.message);
      console.log('👉 Run this manually in the console: copy(JSON.stringify(window.derekPosts))');
    }
  }
})();
