(async function() {

  // ── CONFIG ──────────────────────────────────────
  const TARGET = 20;  // posts to collect
  const SCROLL_WAIT = 7000;  // ms to wait after each scroll (increase if slow connection)
  const MAX_STALE = 25;      // stale rounds before giving up (more = more patient)
  const DEBUG = false;  // set to true to see detailed extraction logs (pass 1, pass 2, etc.)
  // ────────────────────────────────────────────────

  let collected = [], seen = new Set(), staleRounds = 0;
  const wait = ms => new Promise(r => setTimeout(r, ms));
  console.log('🚀 Starting — target: ' + TARGET + ' posts' + (DEBUG ? ' [DEBUG ON]' : ''));

  async function expandSeeMore() {
    [...document.querySelectorAll('[role="button"]')]
      .filter(el => el.innerText?.trim() === 'See more')
      .forEach(btn => { try { btn.click(); } catch(e) {} });
    await wait(1500);
  }

  // Walk up N levels from el to reach the post container (engagement bar confirmed at +4 above body)
  function getPostScope(el, levels = 5) {
    let n = el;
    for (let i = 0; i < levels && n.parentElement; i++) n = n.parentElement;
    return n;
  }

  // Fast static date extraction — checks only timestamp-related elements, not all aria-labels
  function extractDateStatic(art) {
    if (!art) return { date: '', lnk: null };
    const MONTH = /\b(january|february|march|april|may|june|july|august|september|october|november|december|jan|feb|mar|apr|jun|jul|aug|sep|oct|nov|dec)\b/i;
    const DATE_FULL = /\b(20\d{2}|19\d{2})\b|at \d{1,2}:\d{2}/i;

    // 1.  — standard HTML, most reliable if FB uses it
    for (const el of art.querySelectorAll('time[datetime], [datetime]')) {
      const dt = el.getAttribute('datetime') || '';
      if (dt) return { date: new Date(dt).toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric', hour: 'numeric', minute: '2-digit' }), lnk: null };
    }

    // 2. data-utime Unix timestamp
    const uEl = art.querySelector('[data-utime]');
    if (uEl) {
      const ts = parseInt(uEl.getAttribute('data-utime'));
      if (ts) return { date: new Date(ts * 1000).toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }), lnk: null };
    }

    // 3. Timestamp link — exclude comment links, get the post's own header link
    const lnk = art.querySelector('a[href*="/posts/"]:not([href*="comment_id"]),a[href*="story_fbid"]:not([href*="comment_id"]),a[href*="permalink"]:not([href*="comment_id"])');
    if (lnk) {
      // Check every attribute on the link for a date
      for (const attr of lnk.getAttributeNames()) {
        const val = lnk.getAttribute(attr) || '';
        if ((MONTH.test(val) || DATE_FULL.test(val)) && val.length < 100) return { date: val, lnk: null };
      }
      // Check attributes on children of the link
      for (const s of lnk.querySelectorAll('*')) {
        for (const attr of s.getAttributeNames()) {
          const val = s.getAttribute(attr) || '';
          if ((MONTH.test(val) || DATE_FULL.test(val)) && val.length < 100) return { date: val, lnk: null };
        }
        const t = (s.innerText || '').trim();
        if ((MONTH.test(t) || DATE_FULL.test(t)) && t.length < 60) return { date: t, lnk: null };
      }
      // Return the link element for date debug logging — innerText is the relative time
      return { date: lnk.innerText?.trim() || '', lnk };
    }

    // 4. data-store JSON fallback
    for (const el of art.querySelectorAll('[data-store]')) {
      try {
        const ds = JSON.parse(el.getAttribute('data-store') || '{}');
        if (ds.time) return { date: new Date(ds.time * 1000).toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }), lnk: null };
      } catch(e) {}
    }
    return { date: '', lnk: null };
  }

  // Convert Facebook's relative timestamp text to an actual date.
  // Facebook shows "3h", "2d", "Mon", "January 5" etc. on the timestamp link.
  // Since we know the current time when the script runs, we can compute it.
  function resolveRelativeDate(rel) {
    if (!rel) return '';
    const now = new Date();
    const s = rel.trim();

    // "Xm" minutes ago
    let m = s.match(/^(\d+)\s*m$/i);
    if (m) return fmtDate(new Date(now - parseInt(m[1]) * 60000));

    // "Xh" hours ago
    m = s.match(/^(\d+)\s*h$/i);
    if (m) return fmtDate(new Date(now - parseInt(m[1]) * 3600000));

    // "Xd" days ago
    m = s.match(/^(\d+)\s*d$/i);
    if (m) return fmtDate(new Date(now - parseInt(m[1]) * 86400000));

    // "Xw" weeks ago
    m = s.match(/^(\d+)\s*w$/i);
    if (m) return fmtDate(new Date(now - parseInt(m[1]) * 604800000));

    // "Xy" years ago
    m = s.match(/^(\d+)\s*y$/i);
    if (m) return fmtDate(new Date(now - parseInt(m[1]) * 31536000000));

    // "Yesterday"
    if (/^yesterday$/i.test(s)) return fmtDate(new Date(now - 86400000));

    // Day name: "Monday", "Tuesday" etc. — find the most recent past occurrence
    const DAYS = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
    const dayIdx = DAYS.findIndex(d => new RegExp('^' + d, 'i').test(s));
    if (dayIdx >= 0) {
      const diff = (now.getDay() - dayIdx + 7) % 7 || 7;
      return fmtDate(new Date(now - diff * 86400000));
    }

    // Already a formatted date string — return as-is
    return s;
  }

  function fmtDate(d) {
    return d.toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' });
  }

  // Validate that a string is actually a date, not a name or other text
  function isLikelyDate(str) {
    if (!str) return false;
    const s = str.toLowerCase().trim();
    // Should contain a month name or time unit (d/h/m/s/w/y) or be in date format
    const datePatterns = /\b(january|february|march|april|may|june|july|august|september|october|november|december|jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)\b|\d{1,2}[,\s]\s*\d{4}|\d{1,2}\s*(days?|hours?|minutes?|seconds?|weeks?|months?|years?)|\b(yesterday|today|tomorrow)\b/i;
    // Reject if it looks like a person's name (contains "by", no numbers, multiple capital letters)
    if (/^[A-Z][a-z]+\s+[A-Z][a-z]+$/.test(str) && !/\d/.test(str)) return false;
    // Reject single words that are clearly not dates
    if (/^(marketplace|home|feed|share|comment|like|love|post)$/i.test(s)) return false;
    return datePatterns.test(str);
  }

  function parseKM(s) {
    s = (s || '').replace(/,/g, '').trim();
    if (/[Kk]$/.test(s)) return Math.round(parseFloat(s) * 1000);
    if (/[Mm]$/.test(s)) return Math.round(parseFloat(s) * 1000000);
    return parseInt(s) || 0;
  }

  // Returns top-level post body elements (excludes nested shared-post previews)
  function getPostBodies() {
    let nodes = [...document.querySelectorAll('[data-ad-comet-preview="message"]')]
      // Filter out elements that are nested inside another matching element (shared post previews)
      .filter(n => !n.parentElement?.closest('[data-ad-comet-preview="message"]'));
    if (nodes.length === 0)
      nodes = [...document.querySelectorAll('[data-ad-preview="message"]')]
        .filter(n => !n.parentElement?.closest('[data-ad-preview="message"]'));
    if (nodes.length === 0) {
      nodes = [...document.querySelectorAll('[role="article"]')]
        .map(art => art.querySelector('[dir="auto"]'))
        .filter(Boolean);
    }
    return nodes;
  }

  function harvest() {
    for (let body of getPostBodies()) {
      if (collected.length >= TARGET) break;
      [...body.querySelectorAll('[role="button"]')]
        .filter(el => el.innerText?.trim() === 'See more')
        .forEach(btn => { try { btn.click(); } catch(e) {} });

      const artScope = getPostScope(body, 5);
      let postUrl = '', dateLabel = '';

      // Pass 1: Find timestamp link by checking title/aria-label, then text content
      const TS_PAT = /^(\d+\s*[smhdwy]|yesterday|monday|tuesday|wednesday|thursday|friday|saturday|sunday|jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)/i;

      // Try multiple scopes: artScope (5 levels), then wider scope (15 levels), then entire page
      let allLinks = [...artScope.querySelectorAll('a[href]')];
      let tsLnk = null;
      let strategy = '';
      let scopeUsed = 'artScope(5)';

      // Try artScope first (original 5-level scope)
      for (let scopeLevel = 5; scopeLevel <= 30; scopeLevel += 10) {
        tsLnk = null;
        if (scopeLevel === 5) {
          allLinks = [...artScope.querySelectorAll('a[href]')];
          scopeUsed = 'artScope(5)';
        } else if (scopeLevel === 15) {
          allLinks = [...getPostScope(body, 15).querySelectorAll('a[href]')];
          scopeUsed = 'artScope(15)';
        } else {
          allLinks = [...document.querySelectorAll('a[href]')];
          scopeUsed = 'page-wide';
        }

        // Strategy 1a: Look for link with timestamp in title or aria-label
        tsLnk = allLinks.find(a => {
          const attr = (a.getAttribute('title') || a.getAttribute('aria-label') || '').trim();
          return attr.length > 0 && attr.length <= 30 && TS_PAT.test(attr);
        });
        if (tsLnk) { strategy = '1a (attr)'; break; }

        // Strategy 1b: Look for link with timestamp in text (clean whitespace)
        tsLnk = allLinks.find(a => {
          try {
            const t = (a.innerText || '').replace(/[^\x20-\x7E]/g, '').replace(/\s+/g, ' ').trim();
            return t.length > 0 && t.length <= 25 && TS_PAT.test(t);
          } catch(e) { return false; }
        });
        if (tsLnk) { strategy = '1b (text)'; break; }

        // Strategy 1c: Look for link with data-utime attribute (unix timestamp)
        tsLnk = allLinks.find(a => a.getAttribute('data-utime'));
        if (tsLnk) { strategy = '1c (utime)'; break; }

        // Strategy 1d: Look for link containing timestamp patterns in href
        tsLnk = allLinks.find(a => {
          const href = (a.href || '');
          return /\/story\.php|[?&]fbid=|\d{13,}/.test(href);
        });
        if (tsLnk) { strategy = '1d (href)'; break; }
      }


      if (tsLnk) {
        postUrl = tsLnk.href;
        // Try to get date from various sources
        let titleAttr = tsLnk.getAttribute('title') || tsLnk.getAttribute('aria-label') || '';
        const relText = (tsLnk.innerText || '').replace(/\s+/g, ' ').trim();
        const utime = tsLnk.getAttribute('data-utime');

        if (titleAttr && TS_PAT.test(titleAttr)) {
          const candidate = resolveRelativeDate(titleAttr) || titleAttr;
          if (isLikelyDate(candidate)) dateLabel = candidate;
        }
        if (!dateLabel && relText && TS_PAT.test(relText)) {
          const candidate = resolveRelativeDate(relText) || relText;
          if (isLikelyDate(candidate)) dateLabel = candidate;
        }
        if (!dateLabel && utime) {
          const ts = parseInt(utime) * 1000;
          const now = Date.now();
          if (ts > 1420000000000 && ts <= now + 86400000) {
            dateLabel = fmtDate(new Date(ts));
          }
        }
      }

      // Only log if we didn't find a date in any scope (rare now)
      if (!dateLabel && scopeUsed === 'page-wide' && DEBUG) {
        console.log('⚠️ Could not extract date from any link');
      }

      // Pass 2: If no timestamp link found (older posts), search page for dates and find closest match
      if (!dateLabel) {
        const FULL_DATE_RE = /\b(january|february|march|april|may|june|july|august|september|october|november|december)\b.{1,6}\d{1,2}.{0,4}\d{4}/i;
        const now = Date.now();
        let candidates = [];

        // Scan entire page for date signals (data-utime, time[datetime], aria-label/title with dates)
        for (const el of document.querySelectorAll('[data-utime]')) {
          const ts = parseInt(el.getAttribute('data-utime')) * 1000;
          if (ts > 1420000000000 && ts <= now + 86400000) {
            candidates.push({el, date: fmtDate(new Date(ts))});
          }
        }
        for (const el of document.querySelectorAll('time[datetime]')) {
          const val = el.getAttribute('datetime') || '';
          const d = new Date(val);
          if (!isNaN(d) && d.getFullYear() > 2010) {
            candidates.push({el, date: fmtDate(d)});
          }
        }
        for (const el of document.querySelectorAll('[aria-label],[title]')) {
          const v = el.getAttribute('aria-label') || el.getAttribute('title') || '';
          if (FULL_DATE_RE.test(v)) {
            const clean = v.replace(/\s+at\s+\d{1,2}:\d{2}\s*(AM|PM)?/i, '').trim();
            candidates.push({el, date: clean});
          }
        }

        // Find the date candidate whose element is closest to the post body
        if (candidates.length) {
          let closestDist = Infinity, closestDate = '';
          for (const {el, date} of candidates) {
            // Calculate distance: how many steps to walk up from body to find el
            let dist = 0, n = body, found = false;
            while (n && dist <= 20) {
              if (n === el) {
                found = true;
                if (dist < closestDist) { closestDist = dist; closestDate = date; }
                break;
              }
              n = n.parentElement;
              dist++;
            }
          }
          if (closestDate) { dateLabel = closestDate; if (DEBUG) console.log('📅 Pass 2 matched: ' + closestDate + ' (distance: ' + closestDist + ')'); }
          else if (DEBUG) console.log('⚠️ Pass 2 found ' + candidates.length + ' candidates but none were in ancestor chain (max distance 20)');
        } else {
          if (DEBUG) console.log('⚠️ Pass 2 found 0 date candidates on page');
        }
      } else {
        if (DEBUG) console.log('✓ Pass 1 (timestamp link) got date');
      }

      // Pass 3: Try to extract date from post URL itself or any remaining text
      if (!dateLabel) {
        const FULL_DATE_RE = /\b(january|february|march|april|may|june|july|august|september|october|november|december)\b.{1,6}\d{1,2}.{0,4}\d{4}/i;
        // Check if URL itself contains a date
        if (postUrl && FULL_DATE_RE.test(postUrl)) {
          const match = postUrl.match(FULL_DATE_RE);
          if (match) dateLabel = match[0];
        }
        // Look for any text that looks like a date in the post's expanded scope
        if (!dateLabel) {
          const expandedScope = getPostScope(body, 8);
          for (const el of expandedScope.querySelectorAll('*')) {
            if (el.children.length > 0) continue; // Skip containers, look at leaf elements
            const text = (el.innerText || '').trim();
            if (text.length > 50) continue; // Skip long text
            if (FULL_DATE_RE.test(text)) {
              dateLabel = text.match(FULL_DATE_RE)[0];
              if (DEBUG) console.log('📅 Pass 3 found date in text: ' + dateLabel);
              break;
            }
          }
        }
      }

      // Pass 5: find post permalink URL independently of the date search.
      // For recent posts the timestamp link points to the profile, not the post.
      // Look for any link in scope whose href contains a post-specific path segment.
      if (!postUrl) {
        const PERMALINK_RE = /\/(posts|permalink|story_fbid|videos|photos|events)[\/?]/i;
        for (const a of artScope.querySelectorAll('a[href]')) {
          const h = a.href || '';
          if (PERMALINK_RE.test(h) && !h.includes('comment_id')) { postUrl = h; break; }
        }
      }
      // Fallback: use the first non-profile link (avoids profile URL but grabs something unique)
      if (!postUrl) {
        for (const a of artScope.querySelectorAll('a[href]')) {
          const h = a.href || '';
          // Skip profile links, reaction links, empty hrefs
          if (h && !h.includes('/derek.michael.hart') && h.startsWith('https://www.facebook.com')) {
            postUrl = h; break;
          }
        }
      }

      const key = postUrl || body.innerText?.trim().substring(0, 60);
      if (seen.has(key)) continue; seen.add(key);

      // Preserve blank lines — they represent intentional pauses/spacing
      const paras = [...body.querySelectorAll('[dir="auto"]')]
        .map(el => el.innerText?.trimEnd() ?? '');
      const deduped = paras.filter((t, i) => t !== paras[i - 1] || t === '');
      const text = deduped.join('\n\n').replace(/(\n\n){3,}/g, '\n\n\n').trim();
      if (!text || text.length < 30) continue;

      // ── Likes & comments extraction ──────────────────────────────
      let likes = 0, comments = 0;
      // Facebook's engagement bar (reactions, comments) lives OUTSIDE role="article",
      // as a sibling in the parent container. artScope already covers article + siblings.
      const artEl = artScope;

      // Strategy 1: aria-label on the reaction summary button
      // FB currently uses: "75 people reacted" OR "Like: 56 people, Love: 15 people" tooltip style
      for (const el of artEl.querySelectorAll('[aria-label]')) {
        const lbl = (el.getAttribute('aria-label') || '').trim();
        // Per-type tooltip format: "Like: 56 people" (appears in reaction hover tooltip elements)
        const mType = lbl.match(/^(Like|Love|Care|Haha|Wow|Sad|Angry):\s*([\d,.]+[KkMm]?)\s*people/i);
        if (mType) { likes += parseKM(mType[2]); continue; }
        // Aggregate format: "75 people reacted" or "75 reactions"
        if (!likes) {
          const mAgg = lbl.match(/([\d,.]+[KkMm]?)\s*(?:people\s+)?react(?:ed|ions?)?/i);
          if (mAgg) likes = parseKM(mAgg[1]);
        }
        // Comments in aria-label: "8 comments"
        if (!comments) {
          const mCmt = lbl.match(/^([\d,.]+[KkMm]?)\s*[Cc]omments?$/);
          if (mCmt) comments = parseKM(mCmt[1]);
        }
      }

      // Strategy 2a: element whose own text includes "X comments" (e.g. "View 8 comments")
      // Use   handling for non-breaking spaces FB sometimes inserts
      if (!comments) {
        for (const el of artEl.querySelectorAll('a, span, button')) {
          const t = (el.innerText || '').replace(/[ \s]+/g, ' ').trim();
          const m = t.match(/\b([\d,.]+[KkMm]?)\s*[Cc]omments?\b/i);
          if (m) { comments = parseKM(m[1]); break; }
        }
      }

      // Strategy 2b: FB splits number and "comment" into sibling spans —
      // find a leaf number span whose PARENT's full text mentions "comment"
      if (!comments) {
        for (const el of artEl.querySelectorAll('span, a')) {
          const t = (el.innerText || '').trim();
          if (!/^\d[\d,.]*[KkMm]?$/.test(t)) continue; // must be a bare number
          const pText = (el.parentElement?.innerText || '').replace(/[ \s]+/g, ' ').trim();
          if (/\bcomments?\b/i.test(pText) && pText.length < 80) {
            comments = parseKM(t); break;
          }
        }
      }

      // Strategy 2c: scan ALL small elements for any text mentioning comments count
      // Catches "View all 8 comments", "8 Comments", "8\xa0comments" etc.
      if (!comments) {
        for (const el of artEl.querySelectorAll('*')) {
          if (el.children.length > 3) continue;
          const t = (el.innerText || '').replace(/[\s\xa0]+/g, ' ').trim();
          if (t.length > 60) continue;
          const m = t.match(/\b([\d,.]+[KkMm]?)\s*[Cc]omments?\b/) ||
                    t.match(/[Vv]iew\s+(?:all\s+)?([\d,.]+[KkMm]?)\s*[Cc]omments?/);
          if (m) { comments = parseKM(m[1]); break; }
        }
      }

      // Strategy 3: For short threads FB renders comments inline with no count label.
      // Count aria-label="Comment by ..." elements — each visible comment has one.
      if (!comments) {
        comments = artEl.querySelectorAll('[aria-label^="Comment by"]').length;
      }

      const popularity = likes + comments;

      collected.push({ date: dateLabel, url: postUrl, text, likes, comments, popularity });
      console.log('✅ [' + collected.length + '/' + TARGET + '] 📅' + (dateLabel || '(hover pending)') + ' 👍' + likes + ' 💬' + comments + ' — ' + text.replace(/\n/g,' ').substring(0,60) + '...');
    }
  }

  for (let round = 0; collected.length < TARGET; round++) {
    console.log('↓ Scroll #' + (round+1) + ' — ' + collected.length + ' posts so far');
    await expandSeeMore();
    const prev = collected.length;
    await harvest(); await wait(1000); await harvest();
    if (collected.length === prev) {
      staleRounds++;
      console.log('⏳ No new posts this round (' + staleRounds + '/' + MAX_STALE + ' stale)');
      if (staleRounds >= MAX_STALE) { console.log('⚠️ Page exhausted after ' + MAX_STALE + ' stale rounds.'); break; }
    } else staleRounds = 0;
    window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });
    await wait(SCROLL_WAIT);
  }

  if (!collected.length) { console.log('❌ Nothing found.'); return; }

  window.derekPosts = collected;
  console.log('\n✅ DONE — ' + collected.length + ' posts. Copying to clipboard...');
  await wait(800);

  // DevTools copy() is the most reliable method in console context.
  // Using compact JSON (no indentation) keeps the string small and fast.
  let ok = false;
  try {
    copy(JSON.stringify(window.derekPosts));
    ok = true;
    console.log('📋 Copied! Switch to the Import page, click into the JSON box, and press Ctrl+V.');
  } catch(e) {
    console.log('⚠️ copy() failed: ' + e.message);
  }
  if (!ok) {
    // Fallback: navigator.clipboard works on HTTPS (Facebook is HTTPS)
    try {
      await navigator.clipboard.writeText(JSON.stringify(window.derekPosts));
      ok = true;
      console.log('📋 Copied via clipboard API! Switch to Import page and paste (Ctrl+V).');
    } catch(e) {
      console.log('⚠️ Clipboard API failed: ' + e.message);
      console.log('👉 Run this manually in the console: copy(JSON.stringify(window.derekPosts))');
    }
  }
})();
